#include <IRremote.h>

int RECV_PIN = 10;
IRrecv irrecv(RECV_PIN);
decode_results results;
unsigned long key_value = 0;

// Motor 1 (Left) Pins
int M1_In1 = 8;
int M1_In2 = 7;

// Motor 2 (Right) Pins
int M2_In3 = 5;
int M2_In4 = 4;

void setup() {
  pinMode(M1_In1, OUTPUT);
  pinMode(M1_In2, OUTPUT);
  pinMode(M2_In3, OUTPUT);
  pinMode(M2_In4, OUTPUT);

    Serial.begin(9600);
  irrecv.enableIRIn(); // Start the receiver
    irrecv.blink13(true);
}

void loop() {

if (irrecv.decode(&results)){
 
        if (results.value == 0XFFFFFFFF)
        results.value = key_value;

        switch(results.value){
          case 0xFF18E7:
           forward();
           delay(2000);
           stopMotors();
           delay(1000);
          break ;
          case 0xFF10EF:
            turnLeft();
            delay(500);
            stopMotors();
            delay(1000);
            break ;
          case 0xFF38C7:
            stopMotors();
          break ;
          case 0xFF5AA5:
           turnRight();
           delay(500);
           stopMotors();
           delay(1000);
          break ;
          case 0xFF4AB5:
           backward();
           delay(2000);
           stopMotors();
           delay(1000);
          break ;   
        }
        key_value = results.value;
        irrecv.resume(); 
  }

}

void forward() {
  digitalWrite(M1_In1, HIGH);
  digitalWrite(M1_In2, LOW);
  digitalWrite(M2_In3, HIGH);
  digitalWrite(M2_In4, LOW);
}

void backward() {
  digitalWrite(M1_In1, LOW);
  digitalWrite(M1_In2, HIGH);
  digitalWrite(M2_In3, LOW);
  digitalWrite(M2_In4, HIGH);
}

void turnLeft() {
  digitalWrite(M1_In1, LOW);
  digitalWrite(M1_In2, HIGH); // Left backward
  digitalWrite(M2_In3, HIGH); // Right forward
  digitalWrite(M2_In4, LOW);
}

void turnRight() {
  digitalWrite(M1_In1, HIGH); // Left forward
  digitalWrite(M1_In2, LOW);
  digitalWrite(M2_In3, LOW); // Right backward
  digitalWrite(M2_In4, HIGH);
}

void stopMotors() {
  digitalWrite(M1_In1, LOW);
  digitalWrite(M1_In2, LOW);
  digitalWrite(M2_In3, LOW);
  digitalWrite(M2_In4, LOW);
}
